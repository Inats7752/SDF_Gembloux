    <script type="module">

        import { initializeApp } from "https://www.gstatic.com/firebasejs/10.5.0/firebase-app.js";

        import { getFirestore, doc, onSnapshot, setDoc } from "https://www.gstatic.com/firebasejs/10.5.0/firebase-firestore.js";


        /* =======================================================================

           ÉTAPE CRUCIALE : REMPLACEZ CE BLOC PAR VOS PROPRES CLÉS FIREBASE

           ======================================================================= */

        const firebaseConfig = {

            apiKey: "AIzaSyCb1eyEGfXn2FyVHgXgYfrKzmQpeqkvuq8",

            authDomain: "sdf-gembloux.firebaseapp.com",

            projectId: "sdf-gembloux",

            storageBucket: "sdf-gembloux.firebasestorage.app",

            appId: "1:201060005920:web:1664913457482abbba2445"

        };

        /* ======================================================================= */


        // Initialisation de Firebase

        const app = initializeApp(firebaseConfig);

        const dbFirestore = getFirestore(app);

        const docRef = doc(dbFirestore, "sdf_database", "main_data");


        // Base de données par défaut (si Firebase est vide)

        let db = { settings: { maxBeds: 5 }, reservations: {} };

        

        let currentDate = new Date();

        let selectedDateStr = null;

        const todayStr = new Date().toISOString().split('T')[0];


        // ÉCOUTEUR EN TEMPS RÉEL (Met à jour le site dès qu'il y a un changement dans la base)

        onSnapshot(docRef, (docSnap) => {

            if (docSnap.exists()) {

                db = docSnap.data();

            } else {

                setDoc(docRef, db); // Crée le fichier de base si c'est la 1ère fois

            }

            

            document.getElementById('loadingMsg').style.display = 'none';

            if(document.getElementById('adminSection').style.display !== 'block') {

                document.getElementById('clientSection').style.display = 'block';

            }

            

            document.getElementById('maxBedsInput').value = db.settings.maxBeds;

            renderCalendar();

            renderAdminList();

        });


        async function saveDB() {

            await setDoc(docRef, db);

        }


        // --- FONCTIONS DU CALENDRIER (Attachées à "window" pour être utilisables dans le HTML) ---

        window.changeMonth = function(step) {

            currentDate.setMonth(currentDate.getMonth() + step);

            renderCalendar();

        };


        window.selectDate = function(dateStr) {

            selectedDateStr = dateStr;

            renderCalendar(); 

            const dateObj = new Date(dateStr);

            const dateFR = dateObj.toLocaleDateString('fr-FR', { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' });

            document.getElementById('selectedDateDisplay').innerHTML = `Réservation pour le <strong>${dateFR}</strong>`;

            document.getElementById('bookingZone').style.display = 'block';

            document.getElementById('nameInput').focus();

        };


        window.bookBed = async function() {

            const name = document.getElementById('nameInput').value.trim();

            if (!selectedDateStr) return alert("Veuillez choisir une date.");

            if (!name) return alert("Veuillez entrer un nom.");


            if (!db.reservations[selectedDateStr]) db.reservations[selectedDateStr] = [];


            if (db.reservations[selectedDateStr].length >= db.settings.maxBeds) {

                alert("Désolé, quelqu'un vient juste de réserver la dernière place !");

                return;

            }


            db.reservations[selectedDateStr].push(name);

            await saveDB(); // Sauvegarde sur internet

            

            document.getElementById('nameInput').value = '';

            document.getElementById('bookingZone').style.display = 'none';

            selectedDateStr = null;

            alert("Réservation confirmée pour " + name + " !");

        };


        function renderCalendar() {

            const year = currentDate.getFullYear();

            const month = currentDate.getMonth();

            const monthNames = ["Janvier", "Février", "Mars", "Avril", "Mai", "Juin", "Juillet", "Août", "Septembre", "Octobre", "Novembre", "Décembre"];

            document.getElementById('currentMonthDisplay').innerText = `${monthNames[month]} ${year}`;


            const calendarDays = document.getElementById('calendarDays');

            calendarDays.innerHTML = '';


            let firstDay = new Date(year, month, 1).getDay() - 1;

            if (firstDay === -1) firstDay = 6; 

            const daysInMonth = new Date(year, month + 1, 0).getDate();


            for (let i = 0; i < firstDay; i++) {

                const emptyDiv = document.createElement('div');

                emptyDiv.style.visibility = 'hidden';

                calendarDays.appendChild(emptyDiv);

            }


            for (let i = 1; i <= daysInMonth; i++) {

                const dayDiv = document.createElement('div');

                dayDiv.className = 'calendar-day';

                const dStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(i).padStart(2, '0')}`;

                

                const bookedCount = db.reservations[dStr] ? db.reservations[dStr].length : 0;

                const remaining = db.settings.maxBeds - bookedCount;


                dayDiv.innerHTML = `<span class="day-number">${i}</span><span class="day-spots">${remaining} dispo</span>`;


                if (dStr < todayStr) {

                    dayDiv.classList.add('disabled');

                    dayDiv.querySelector('.day-spots').innerText = "Passé";

                } else if (remaining <= 0) {

                    dayDiv.classList.add('full');

                    dayDiv.querySelector('.day-spots').innerText = "Complet";

                } else {

                    if (dStr === selectedDateStr) dayDiv.classList.add('selected');

                    dayDiv.onclick = () => window.selectDate(dStr);

                }

                calendarDays.appendChild(dayDiv);

            }

        }


        // --- FONCTIONS ADMIN ---

        window.toggleAdmin = function() {

            const clientSec = document.getElementById('clientSection');

            const adminSec = document.getElementById('adminSection');


            if (adminSec.style.display === 'none' || adminSec.style.display === '') {

                const pwd = prompt("Mot de passe Admin ? (Tapez 'admin')");

                if (pwd === 'admin') {

                    clientSec.style.display = 'none';

                    adminSec.style.display = 'block';

                } else { alert("Mot de passe incorrect."); }

            } else {

                clientSec.style.display = 'block';

                adminSec.style.display = 'none';

                selectedDateStr = null;

                document.getElementById('bookingZone').style.display = 'none';

            }

        };


        window.updateSettings = async function() {

            const newMax = parseInt(document.getElementById('maxBedsInput').value);

            if (newMax > 0) {

                db.settings.maxBeds = newMax;

                await saveDB();

                alert("Nombre de lits mis à jour pour tout le monde !");

            }

        };


        function renderAdminList() {

            const container = document.getElementById('adminReservationsList');

            container.innerHTML = '';

            const dates = Object.keys(db.reservations).sort();

            

            if(dates.length === 0) {

                container.innerHTML = '<p>Aucune réservation pour le moment.</p>';

                return;

            }


            dates.forEach(date => {

                if (db.reservations[date].length > 0) {

                    const group = document.createElement('div');

                    group.className = 'admin-date-group';

                    const dateObj = new Date(date);

                    const dateFR = dateObj.toLocaleDateString('fr-FR', { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' });


                    let html = `<div class="admin-date-title">${dateFR} (${db.reservations[date].length}/${db.settings.maxBeds} lits)</div><ul class="admin-list">`;

                    db.reservations[date].forEach((name, index) => {

                        html += `<li><span>${name}</span> <button class="btn btn-small btn-danger" onclick="window.deleteBooking('${date}', ${index})">Supprimer</button></li>`;

                    });

                    html += `</ul>`;

                    group.innerHTML = html;

                    container.appendChild(group);

                }

            });

        }


        window.deleteBooking = async function(date, index) {

            if(confirm("Annuler cette réservation ?")) {

                db.reservations[date].splice(index, 1);

                if(db.reservations[date].length === 0) delete db.reservations[date];

                await saveDB();

            }

        };
